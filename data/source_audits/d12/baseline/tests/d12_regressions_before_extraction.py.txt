@pytest.mark.parametrize("player", ["player-a", "player-b"])
@pytest.mark.parametrize(
    ("attached", "selected"), [(False, "leader"), (True, "leader"), (True, "support")]
)
def test_d12_epic_model_owned_melee_restore_replay_and_expiry(
    player: str, attached: bool, selected: str
) -> None:
    from tests.psychic_modifier_helpers import pending_request

    session = epic_scene(
        player=player,
        attached=attached,
        support=attached,
        armed_bodyguard=attached,
        pause_attacks=True,
    )
    request = epic_request(session, player=player)
    state = session.lifecycle.state
    assert state is not None
    unit = epic_unit_from_state(state, player=player)
    model = next(
        model
        for component in unit.components
        if component.unit.unit_instance_id.endswith(":" + selected)
        for model in component.unit.own_models
    )
    saved = session.to_persistence_payload()
    restored, fork = epic_checkpoint(session)
    submit_epic(restored, request, epic_proposal(request, model_id=model.model_instance_id))
    assert session.to_persistence_payload() == fork.to_persistence_payload() == saved
    session = restored
    state = session.lifecycle.state
    assert state is not None
    assert state.command_point_total(player) == 1
    assert len(state.stratagem_use_records) == 1
    effect = state.persisting_effects[0]
    assert effect.target_unit_instance_ids == (unit.unit_instance_id,)
    assert effect.expiration == EffectExpiration.end_phase(
        battle_round=1, phase=BattlePhase.FIGHT, player_id=player
    )
    session, _ = epic_checkpoint(session)
    request = pending_request(session)
    assert request.decision_type == "submit_melee_declaration"
    payload = cast(dict[str, JsonValue], request.payload)
    melee = cast(dict[str, JsonValue], payload["proposal_request"])
    weapons = cast(list[dict[str, JsonValue]], melee["available_weapons"])
    seen: set[str] = set()
    for weapon in weapons:
        owner = weapon["model_instance_id"]
        assert isinstance(owner, str)
        if weapon["weapon_ability_selection_context"] is None:
            assert weapon["engaged_target_unit_instance_ids"] == []
            assert owner != model.model_instance_id
            continue
        context = cast(dict[str, JsonValue], weapon["weapon_ability_selection_context"])
        for target in cast(list[dict[str, JsonValue]], context["target_profiles"]):
            profile = cast(dict[str, JsonValue], target["weapon_profile"])
            assert ("Precision" in cast(list[str], profile["keywords"])) is (
                owner == model.model_instance_id
            )
            seen.add(owner)
    assert model.model_instance_id in seen
    if attached:
        assert any("bodyguard" in owner for owner in seen)
        assert any("support" in owner for owner in seen)
        assert any("leader" in owner for owner in seen)
    submit_all_melee(session, request)
    state = session.lifecycle.state
    assert state is not None
    assert state.fight_phase_state is not None
    sequence = state.fight_phase_state.attack_sequence
    assert sequence is not None
    pools = sequence.attack_pools
    assert pools
    for pool in pools:
        assert (WeaponKeyword.PRECISION in pool.weapon_profile.keywords) is (
            pool.attacker_model_instance_id == model.model_instance_id
        )
    session, _ = epic_checkpoint(session)
    finish_epic_fight(session, unit_id=unit.unit_instance_id)
    session, _ = epic_checkpoint(session)
    for _ in range(90):
        state = session.lifecycle.state
        assert state is not None
        if state.current_battle_phase is not BattlePhase.FIGHT:
            break
        request = pending_request(session)
        if request.decision_type == "submit_melee_declaration":
            submit_all_melee(session, request)
        else:
            from tests.fight_completion_helpers import submit_completion_request

            submit_completion_request(session)
    else:
        raise AssertionError("D12 did not complete the ordinary phase.")
    assert effect.effect_id not in {current.effect_id for current in state.persisting_effects}
    epic_checkpoint(session)


@pytest.mark.parametrize("player", ["player-a", "player-b"])
@pytest.mark.parametrize(
    "invalid", ["bodyguard", "friendly", "enemy", "unknown", "boolean", "component"]
)
def test_d12_epic_invalid_model_choice_is_atomic_and_exact_request_retries(
    player: str, invalid: str
) -> None:
    session = epic_scene(player=player, support=True)
    request = epic_request(session, player=player)
    state = session.lifecycle.state
    assert state is not None
    unit = epic_unit_from_state(state, player=player)
    leader = next(
        model
        for component in unit.components
        if component.role == "leader"
        for model in component.unit.own_models
    )
    if invalid == "bodyguard":
        model_id: JsonValue = unit.components[0].unit.own_models[0].model_instance_id
    elif invalid in {"friendly", "enemy"}:
        model_id = next(
            model.model_instance_id
            for army in state.army_definitions
            for physical in army.units
            if physical.unit_instance_id.endswith(":foreign" if invalid == "friendly" else ":enemy")
            for model in physical.own_models
        )
    elif invalid == "unknown":
        model_id = "unadvertised-foreign-model"
    elif invalid == "boolean":
        model_id = True
    else:
        model_id = leader.model_instance_id
    proposal = epic_proposal(request, model_id=model_id)
    if invalid == "component":
        assert proposal.target_binding is not None
        proposal = proposal.with_binding(
            replace(
                proposal.target_binding,
                target_unit_instance_id=unit.components[0].unit.unit_instance_id,
            ),
            effect_selection=proposal.effect_selection,
        )
    saved = session.to_persistence_payload()
    status = session.submit_parameterized_payload(
        request_id=request.request_id,
        result_id=request.request_id + ":d12-invalid",
        payload=validate_json_value({"proposal": proposal.to_payload()}),
    )
    assert status.status_kind is LifecycleStatusKind.INVALID
    if invalid == "bodyguard":
        assert status.payload == {"invalid_reason": "epic_challenge_model_not_character"}
    assert session.to_persistence_payload() == saved
    assert session.lifecycle.decision_controller.queue.peek_next() == request
    submit_epic(session, request, epic_proposal(request, model_id=leader.model_instance_id))
    assert state.command_point_total(player) == 1
    assert len(state.stratagem_use_records) == 1
    assert len(state.persisting_effects) == 1
    epic_checkpoint(session)

@pytest.mark.parametrize("player", ["player-a", "player-b"])
@pytest.mark.parametrize("control", ["no_cp", "noncharacter", "decline"])
def test_d12_epic_preserves_ordinary_and_optional_controls(player: str, control: str) -> None:
    from tests.fight_completion_helpers import submit_completion_request
    from tests.psychic_modifier_helpers import pending_request

    from warhammer40k_core.engine.stratagems import stratagem_decline_payload

    session = epic_scene(
        player=player,
        attached=control != "no_cp",
        cp=0 if control == "no_cp" else 2,
        noncharacter=control == "noncharacter",
    )
    if control == "decline":
        request = epic_request(session, player=player)
        status = session.submit_parameterized_payload(
            request_id=request.request_id,
            result_id=request.request_id + ":decline",
            payload=stratagem_decline_payload(),
        )
        assert status.status_kind is not LifecycleStatusKind.INVALID
    for _ in range(60):
        request = pending_request(session)
        if request.decision_type == "submit_melee_declaration" and request.actor_id == player:
            break
        assert request.decision_type != "submit_stratagem_target_proposal", request
        submit_completion_request(session)
    else:
        raise AssertionError("Ordinary control did not reach native melee.")
    state = session.lifecycle.state
    assert state is not None
    assert state.stratagem_use_records == []
    assert state.persisting_effects == []
    assert state.command_point_total(player) == (0 if control == "no_cp" else 2)
    melee = cast(
        dict[str, JsonValue], cast(dict[str, JsonValue], request.payload)["proposal_request"]
    )
    for row in cast(list[dict[str, JsonValue]], melee["available_weapons"]):
        if row["weapon_ability_selection_context"] is None:
            assert row["engaged_target_unit_instance_ids"] == []
            continue
        context = cast(dict[str, JsonValue], row["weapon_ability_selection_context"])
        for target in cast(list[dict[str, JsonValue]], context["target_profiles"]):
            profile = cast(dict[str, JsonValue], target["weapon_profile"])
            assert "Precision" not in cast(list[str], profile["keywords"])
    submit_all_melee(session, request)
    epic_checkpoint(session)


@pytest.mark.parametrize("player", ["player-a", "player-b"])
def test_d12_native_split_epic_keeps_physical_model_ownership(player: str) -> None:
    from tests.epic_challenge_split_helpers import native_split_epic_request

    from warhammer40k_core.engine.rules_units import rules_unit_view_by_id

    session, request = native_split_epic_request(player=player)
    state = session.lifecycle.state
    assert state is not None
    army = next(army for army in state.army_definitions if army.player_id == player)
    assert len(army.unit_splits) == 1
    split = army.unit_splits[0]
    assert split.used_balanced_fallback
    assert len(split.model_ids(0)) == len(split.model_ids(1)) == 3
    leader_id = next(
        model.model_instance_id
        for unit in split.source_units
        for model in unit.own_models
        if "CHARACTER" in model.keywords
    )
    selected = rules_unit_view_by_id(
        state=state, unit_instance_id=state.unit_instance_id_for_model(leader_id)
    )
    assert ":split:" in selected.unit_instance_id
    assert selected.model_by_id(leader_id).model_instance_id == leader_id
    assert "CHARACTER" in selected.model_by_id(leader_id).keywords
    outsider = next(
        model_id
        for model_id in split.model_ids(0)
        if model_id not in {m.model_instance_id for m in selected.own_models}
    )
    saved = session.to_persistence_payload()
    status = session.submit_parameterized_payload(
        request_id=request.request_id,
        result_id=request.request_id + ":wrong-successor",
        payload=validate_json_value(
            {"proposal": epic_proposal(request, model_id=outsider).to_payload()}
        ),
    )
    assert status.status_kind is LifecycleStatusKind.INVALID
    assert status.payload == {"invalid_reason": "epic_challenge_model_not_in_unit"}
    assert session.to_persistence_payload() == saved
    cp = state.command_point_total(player)
    submit_epic(session, request, epic_proposal(request, model_id=leader_id))
    assert state.command_point_total(player) == cp - 1
    assert state.stratagem_use_records[-1].targeted_unit_instance_ids == (
        selected.unit_instance_id,
    )
    restored, fork = epic_checkpoint(session)
    saved = session.to_persistence_payload()
    finish_epic_fight(restored, unit_id=selected.unit_instance_id)
    assert session.to_persistence_payload() == fork.to_persistence_payload() == saved
    epic_checkpoint(restored)
