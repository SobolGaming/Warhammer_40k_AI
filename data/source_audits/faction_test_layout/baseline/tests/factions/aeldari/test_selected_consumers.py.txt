def test_rangers_infiltrators_and_ranged_invulnerable_queries_are_model_scoped(
    faction_definitions: Section02Definitions,
) -> None:
    d = faction_definitions
    s = scene(
        d,
        (selection(d, RANGERS, "rangers"), selection(d, ELDRAD, "eldrad")),
        (selection(d, LORD, "enemy"),),
    )
    rangers = s.units["own:rangers"]
    assert rules_unit_has_infiltrators(
        state=s.state,
        view=rules_unit_view_by_id(state=s.state, unit_instance_id=rangers.unit_instance_id),
    )
    assert not rules_unit_has_infiltrators(
        state=s.state, view=rules_unit_view_by_id(state=s.state, unit_instance_id="own:eldrad")
    )
    bundle = s.session.lifecycle._require_runtime_content_bundle()  # pyright: ignore[reportPrivateUsage]
    ranged = d.weapon(f"{LORD}:bolt-pistol")
    melee = d.weapon(f"{LORD}:phoenix-power-spear")
    context = SaveOptionModifierContext(
        s.state,
        rangers.unit_instance_id,
        (),
        source_phase=BattlePhase.SHOOTING,
        attacking_unit_instance_id="foe:enemy",
        attacker_model_instance_id=s.units["foe:enemy"].own_model_ids()[0],
        weapon_profile=ranged,
        allocated_model_instance_id=rangers.own_model_ids()[0],
    )
    for mid in rangers.own_model_ids():
        options = bundle.runtime_modifier_registry.modified_save_options(
            replace(context, allocated_model_instance_id=mid)
        )
        assert [(o.save_kind, o.target_number) for o in options] == [(SaveKind.INVULNERABLE, 5)]
    assert (
        bundle.runtime_modifier_registry.modified_save_options(
            replace(context, weapon_profile=melee, source_phase=BattlePhase.FIGHT)
        )
        == ()
    )
    assert (
        bundle.runtime_modifier_registry.modified_save_options(
            replace(context, allocated_model_instance_id=s.units["own:eldrad"].own_model_ids()[0])
        )
        == ()
    )


def test_ranger_stealth_changes_actual_native_shooting_skill(
    faction_definitions: Section02Definitions,
) -> None:
    d = faction_definitions
    s = scene(
        d,
        (selection(d, RANGERS, "rangers"),),
        (selection(d, LORD, "shooter"), selection(d, LORD, "spare")),
        phase=BattlePhase.SHOOTING,
        active="opponent",
    )
    locate(s, "foe:shooter", (Pose.at(22, 10),))
    anchor(s)
    request(s.session.advance_until_decision_or_terminal(), "select_shooting_unit")
    r = shooting_request(s.session, "foe:shooter")
    rows = weapon_rows(r)
    rows = [row for row in rows if row["wargear_id"] == f"{LORD}:bolt-pistol"]
    assert len(rows) == 1
    status = s.session.submit_parameterized_payload(
        request_id=r.request_id,
        payload=shooting_payload(r, rows, ("own:rangers",)),
        result_id="rangers:incoming-shot",
    )
    assert status.status_kind is not LifecycleStatusKind.INVALID, status
    shooting = s.state.shooting_phase_state
    assert shooting is not None
    assert len(shooting.attack_pools) == 1
    pool = shooting.attack_pools[0]
    bundle = s.session.lifecycle._require_runtime_content_bundle()  # pyright: ignore[reportPrivateUsage]
    modifiers = attack_modifier_snapshots(
        state=s.state,
        pool=pool,
        source_phase=BattlePhase.SHOOTING,
        runtime_modifier_registry=bundle.runtime_modifier_registry,
    )
    assert "stealth" in pool.targeting_rule_ids
    assert sum(m.modifier.operand for m in modifiers if m.kind == "skill") == 1
    finish_attacks(s.session)
    steps = events(s.session, "attack_sequence_step")
    hits = [cast(dict[str, Any], e["payload"]) for e in steps if e["step"] == "hit"]
    assert len(hits) == 1
    assert hits[0]["target_number"] == pool.weapon_profile.skill.final + 1
    assert events(s.session, "attack_sequence_completed")


def test_rangers_native_enemy_move_reaction_restores_moves_once_and_replays(
    faction_definitions: Section02Definitions,
) -> None:
    d = faction_definitions
    s = scene(
        d,
        (selection(d, RANGERS, "rangers"),),
        (selection(d, LORD, "mover"), selection(d, LORD, "other")),
        active="opponent",
    )
    locate(s, "foe:mover", (Pose.at(22, 10),))
    locate(s, "foe:other", (Pose.at(22, 14),))
    anchor(s)
    request(s.session.advance_until_decision_or_terminal(), "select_movement_unit")
    r = request(translate(s, "foe:mover", 0.25), "select_triggered_movement")
    assert r.actor_id == "owner"
    assert {o.option_id for o in r.options} == {
        "decline_triggered_movement",
        "triggered:own:rangers",
    }
    saved = json.loads(json.dumps(s.session.to_persistence_payload()))
    restored = LocalGameSession.from_persistence_payload(saved)
    assert restored.to_persistence_payload() == saved
    for current in (s.session, restored):
        pending = current.lifecycle.pending_decision_request()
        assert pending is not None
        r = pending
        r = request(
            current.submit_option(
                request_id=r.request_id,
                option_id="triggered:own:rangers",
                result_id="rangers:react",
            ),
            "submit_movement_proposal",
        )
        before = current.lifecycle.to_payload()
        invalid = current.submit_parameterized_payload(
            request_id=r.request_id, payload={}, result_id="rangers:malformed"
        )
        assert invalid.status_kind is LifecycleStatusKind.INVALID
        assert current.lifecycle.to_payload() == before
        assert current.lifecycle.state is not None
        branch = Scene(current, current.lifecycle.state, s.units)
        board = branch.state.battlefield_state
        assert board is not None
        starts = board.unit_placement_by_id("own:rangers")
        status = submit_translation(branch, r, "own:rangers", -0.25)
        assert status.status_kind is not LifecycleStatusKind.INVALID, status
        assert branch.state.battlefield_state is not None
        ends = branch.state.battlefield_state.unit_placement_by_id("own:rangers")
        assert [p.pose.position.x for p in ends.model_placements] == [
            p.pose.position.x - 0.25 for p in starts.model_placements
        ]
        request(status, "select_movement_unit")
        second = request(translate(branch, "foe:other", 0.25), "select_triggered_movement")
        assert {o.option_id for o in second.options} == {"decline_triggered_movement"}
        current.submit_option(
            request_id=second.request_id,
            option_id="decline_triggered_movement",
            result_id="rangers:already-moved",
        )
        assert len(events(current, "triggered_movement_resolved")) == 1
        assert (
            ReplayRunner.from_payload(current.replay_artifact(artifact_id="rangers-reactive"))
            .run()
            .reproduced_exactly
        )
    assert s.session.lifecycle.to_payload() == restored.lifecycle.to_payload()


@pytest.mark.parametrize(
    ("presence", "active"),
    [
        ("battlefield", "owner"),
        ("capped", "owner"),
        ("reserves", "owner"),
        ("destroyed", "owner"),
        ("battlefield", "opponent"),
    ],
)
def test_diviner_cp_gain_requires_own_command_phase_and_physical_presence(
    faction_definitions: Section02Definitions,
    presence: str,
    active: str,
) -> None:
    from warhammer40k_core.engine.command_points import CommandPointSourceKind

    d = faction_definitions
    s = scene(
        d,
        (selection(d, ELDRAD, "eldrad"), selection(d, AVENGERS, "body")),
        (selection(d, LORD, "enemy"),),
        phase=BattlePhase.COMMAND,
        active=active,
    )
    unit = s.units["own:eldrad"]
    if presence == "capped":
        s.state.gain_command_points(
            player_id="owner",
            amount=1,
            source_id="constructed:batch01:prior-other-gain",
            source_kind=CommandPointSourceKind.OTHER,
        )
    if presence == "reserves":
        s.state.record_reserve_state(
            ReserveState.declared_before_battle(
                player_id="owner",
                unit_instance_id=unit.unit_instance_id,
                reserve_kind=ReserveKind.STRATEGIC_RESERVES,
            )
        )
        assert s.state.battlefield_state is not None
        s.state.replace_battlefield_state(
            s.state.battlefield_state.without_unit_placement(unit.unit_instance_id)
        )
    elif presence == "destroyed":
        casualty_checkpoint(s, unit.own_model_ids())
    status = s.session.advance_until_decision_or_terminal()
    for i in range(12):
        if (
            status.decision_request is None
            or status.decision_request.decision_type != "resolve_sequencing_order"
        ):
            break
        r = status.decision_request
        status = s.session.submit_option(
            request_id=r.request_id,
            option_id=r.options[0].option_id,
            result_id=f"diviner:sequence:{i}",
        )
    eligible = presence in {"battlefield", "capped"} and active == "owner"
    gains = events(s.session, "catalog_ir_command_point_phase_gain_resolved")
    assert len(gains) == int(eligible)
    assert s.state.command_point_total("owner") == 1 + int(eligible)
    assert s.state.command_point_total("opponent") == 1
    transactions = s.state.command_point_ledger_for_player("owner").transactions
    assert (
        sum(
            t.amount
            for t in transactions
            if t.source_kind is CommandPointSourceKind.COMMAND_PHASE_START
        )
        == 1
    )
    assert sum(
        t.amount for t in transactions if t.source_kind is CommandPointSourceKind.OTHER
    ) == int(eligible)
    if eligible:
        cp_result = cast(dict[str, Any], gains[0]["command_point_result"])
        assert cp_result["status"] == ("capped" if presence == "capped" else "applied")
        assert cp_result["applied_amount"] == int(presence != "capped")


def test_doom_selects_only_eligible_target_modifies_wounds_then_expires(
    faction_definitions: Section02Definitions,
) -> None:
    d = faction_definitions
    s = scene(
        d,
        (selection(d, ELDRAD, "eldrad"), selection(d, AVENGERS, "body")),
        (selection(d, LORD, "near"), selection(d, LORD, "far")),
    )
    locate(s, "foe:near", (Pose.at(20, 10),))
    locate(s, "foe:far", (Pose.at(45, 10),))
    anchor(s)
    own_ids = tuple(
        u.unit_instance_id
        for a in s.state.army_definitions
        if a.player_id == "owner"
        for u in a.units
    )
    s.state.movement_phase_state = MovementPhaseState(
        1, "owner", move_units_completed=True, selected_unit_ids=own_ids, moved_unit_ids=own_ids
    )
    r = request(
        s.session.advance_until_decision_or_terminal(), "select_catalog_movement_end_target_effect"
    )
    payload = cast(dict[str, Any], r.payload)
    assert payload["available_target_unit_instance_ids"] == ["foe:near"]
    valid = DecisionResult.for_request(
        request=r, result_id="doom:valid", selected_option_id=r.options[0].option_id
    )
    bad_payload = json.loads(json.dumps(valid.payload))
    bad_payload["selected_catalog_target_effect"]["target_unit_instance_id"] = "foe:far"
    before = s.session.lifecycle.to_payload()
    assert (
        s.session.lifecycle.submit_decision(replace(valid, payload=bad_payload)).status_kind
        is LifecycleStatusKind.INVALID
    )
    assert s.session.lifecycle.to_payload() == before
    status = s.session.submit_option(
        request_id=r.request_id, option_id=r.options[0].option_id, result_id="doom:near"
    )
    assert status.status_kind is not LifecycleStatusKind.INVALID, status
    effects = [
        e for e in s.state.persisting_effects if e.source_rule_id == payload["source_rule_id"]
    ]
    assert len(effects) == 1
    assert effects[0].target_unit_instance_ids == ("own:body", "own:eldrad")
    registry = s.session.lifecycle._require_runtime_content_bundle().runtime_modifier_registry  # pyright: ignore[reportPrivateUsage]
    body = s.units["own:body"]
    ctx = WoundRollModifierContext(
        s.state,
        BattlePhase.SHOOTING,
        body.unit_instance_id,
        body.own_model_ids()[0],
        "foe:near",
        d.weapon(f"{AVENGERS}:avenger-shuriken-catapult"),
        4,
        4,
    )
    assert registry.wound_roll_modifier(ctx) == 1
    assert registry.wound_roll_modifier(replace(ctx, target_unit_instance_id="foe:far")) == 0
    saved = s.session.to_persistence_payload()
    assert LocalGameSession.from_persistence_payload(saved).to_persistence_payload() == saved
    assert (
        ReplayRunner.from_payload(s.session.replay_artifact(artifact_id="doom"))
        .run()
        .reproduced_exactly
    )
    s.state.expire_persisting_effects_at_boundary(
        EffectExpirationBoundary.phase_start(
            battle_round=2, phase=BattlePhaseKind.COMMAND, player_id="owner"
        )
    )
    assert registry.wound_roll_modifier(ctx) == 0

def _battle_focus_spends(s: Scene) -> list[dict[str, Any]]:
    return [
        cast(dict[str, Any], effect.effect_payload)
        for effect in s.state.persisting_effects
        if isinstance(effect.effect_payload, dict)
        and effect.effect_payload.get("effect_kind") == "aeldari_battle_focus_token_spent"
    ]


def _pending(s: Scene) -> DecisionRequest:
    pending = s.session.lifecycle.pending_decision_request()
    assert pending is not None
    return pending


def _choice(s: Scene, option: str) -> LifecycleStatus:
    r = _pending(s)
    return s.session.submit_option(
        request_id=r.request_id, option_id=option, result_id=f"{r.request_id}:aeldari"
    )


def _fall_back(s: Scene, unit_id: str) -> LifecycleStatus:
    from warhammer40k_core.engine.event_log import validate_json_value
    from warhammer40k_core.engine.movement_proposals import (
        MovementProposalPayload,
        MovementProposalRequest,
    )
    from warhammer40k_core.geometry.pathing import PathWitness

    _choice(s, unit_id)
    r = request(_choice(s, "fall_back:ordered_retreat"), "submit_movement_proposal")
    context = MovementProposalRequest.from_decision_request_payload(r.payload)
    assert context.movement_phase_action is not None
    assert s.state.battlefield_state is not None
    placements = s.state.battlefield_state.unit_placement_by_id(unit_id).model_placements
    witness = PathWitness.for_paths(
        tuple(
            (p.model_instance_id, (p.pose, Pose.at(p.pose.position.x + 4, p.pose.position.y)))
            for p in placements
        )
    )
    payload = MovementProposalPayload(
        r.request_id,
        context.proposal_kind,
        unit_id,
        context.movement_phase_action,
        witness,
        movement_mode="fall_back",
        fall_back_mode="ordered_retreat",
    )
    return s.session.submit_parameterized_payload(
        request_id=r.request_id,
        payload=validate_json_value(payload.to_payload()),
        result_id=f"{r.request_id}:retreat",
    )


def _assert_normal_reaction(s: Scene, r: DecisionRequest, maneuver: str) -> float:
    assert r.actor_id == "owner"
    assert isinstance(r.payload, dict)
    descriptor = cast(dict[str, Any], r.payload["descriptor"])
    assert descriptor["movement_kind"] == "triggered"
    assert descriptor["movement_mode"] == "normal"
    assert {o.option_id for o in r.options} == {
        "decline_triggered_movement",
        "triggered:own:reactor",
    }
    eligible = cast(list[dict[str, Any]], r.payload["eligible_units"])
    assert len(eligible) == 1
    assert eligible[0]["hook_id"] == f"warhammer_40000_11th:aeldari:army_rule:{maneuver}"
    assert eligible[0]["unit_instance_id"] == "own:reactor"
    assert eligible[0]["decision_effect_payload"]["maneuver"] == maneuver
    assert eligible[0]["decision_effect_payload"]["battle_focus_token_cost"] == 1
    kind = (
        "movement_end_surge_triggered"
        if maneuver == "opportunity_seized"
        else "shooting_end_surge_triggered"
    )
    trigger = events(s.session, kind)[-1]
    assert trigger["max_distance_bonus_inches"] == 1
    roll = cast(dict[str, Any], trigger["surge_distance_roll"])
    assert descriptor["max_distance_inches"] == roll["current_total"] + 1
    return cast(float, descriptor["max_distance_inches"])


def _accept_normal_reaction(s: Scene, *, recover: bool = False) -> None:
    prior_count = len(_battle_focus_spends(s))
    r = request(_choice(s, "triggered:own:reactor"), "submit_movement_proposal")
    restored: LocalGameSession | None = None
    if recover:
        saved = json.loads(json.dumps(s.session.to_persistence_payload()))
        restored = LocalGameSession.from_persistence_payload(saved)
        assert restored.to_persistence_payload() == saved
    assert s.state.battlefield_state is not None
    before = s.state.battlefield_state.unit_placement_by_id("own:reactor")
    spends = list(_battle_focus_spends(s))
    # A well-formed over-distance path cannot move a model or spend a second token.
    invalid = submit_translation(s, r, "own:reactor", -8)
    assert invalid.status_kind is LifecycleStatusKind.INVALID
    assert s.state.battlefield_state.unit_placement_by_id("own:reactor") == before
    assert _battle_focus_spends(s) == spends
    r = _pending(s)
    # Moving away, and less than the rolled allowance, is a legal Normal move.
    status = submit_translation(s, r, "own:reactor", -0.5)
    assert status.status_kind is not LifecycleStatusKind.INVALID, status
    after = s.state.battlefield_state.unit_placement_by_id("own:reactor")
    assert [p.pose.position.x for p in after.model_placements] == [
        p.pose.position.x - 0.5 for p in before.model_placements
    ]
    resolved = events(s.session, "triggered_movement_resolved")
    assert len(resolved) == 1
    assert resolved[0]["triggered_movement_kind"] == "triggered"
    assert len(_battle_focus_spends(s)) == prior_count + 1
    assert _battle_focus_spends(s)[-1]["battle_focus_token_cost"] == 1
    if recover:
        assert restored is not None
        assert restored.lifecycle.state is not None
        branch = Scene(restored, restored.lifecycle.state, {})
        pending = restored.lifecycle.pending_decision_request()
        assert pending is not None
        invalid = submit_translation(branch, pending, "own:reactor", -8)
        assert invalid.status_kind is LifecycleStatusKind.INVALID
        pending = restored.lifecycle.pending_decision_request()
        assert pending is not None
        submit_translation(branch, pending, "own:reactor", -0.5)
        assert restored.lifecycle.to_payload() == s.session.lifecycle.to_payload()
        assert (
            ReplayRunner.from_payload(s.session.replay_artifact(artifact_id="aeldari-normal"))
            .run()
            .reproduced_exactly
        )


def test_opportunity_seized_native_fall_back_decline_then_normal_move(
    faction_definitions: Section02Definitions,
) -> None:
    d = faction_definitions
    s = scene(
        d,
        (
            selection(d, ELDRAD, "reactor"),
            selection(d, AVENGERS, "other"),
            selection(d, AVENGERS, "declined"),
        ),
        tuple(selection(d, LORD, name) for name in ("first", "second", "third", "spare")),
        active="opponent",
    )
    locate(s, "own:other", tuple(Pose.at(10, 22 + i * 1.5) for i in range(5)))
    locate(s, "own:declined", tuple(Pose.at(10, 34 + i * 1.5) for i in range(5)))
    locate(s, "foe:first", (Pose.at(11.5, 34),))
    locate(s, "foe:second", (Pose.at(11.5, 10),))
    locate(s, "foe:third", (Pose.at(11.5, 22),))
    anchor(s)
    request(s.session.advance_until_decision_or_terminal(), "select_movement_unit")
    r = request(_fall_back(s, "foe:first"), "select_triggered_movement")
    assert {o.option_id for o in r.options} == {
        "decline_triggered_movement",
        "triggered:own:declined",
    }
    assert s.state.battlefield_state is not None
    before = s.state.battlefield_state
    request(_choice(s, "decline_triggered_movement"), "select_movement_unit")
    assert s.state.battlefield_state == before
    assert _battle_focus_spends(s) == []
    r = request(_fall_back(s, "foe:second"), "select_triggered_movement")
    _assert_normal_reaction(s, r, "opportunity_seized")
    _accept_normal_reaction(s, recover=True)
    # A fresh, otherwise eligible unit cannot repeat this manoeuvre this phase.
    r = request(_fall_back(s, "foe:third"), "select_movement_unit")
    assert r.actor_id == "opponent"
    assert len(_battle_focus_spends(s)) == 1

        d,
        (selection(d, ELDRAD, "reactor"), selection(d, AVENGERS, "other")),
        (selection(d, LORD, "shooter"), selection(d, LORD, "spare")),
        active="opponent",
        phase=BattlePhase.SHOOTING,
        seed=seed,
    )
    locate(s, "foe:shooter", (Pose.at(18, 10),))
    anchor(s)
    request(s.session.advance_until_decision_or_terminal(), "select_shooting_unit")
    r = shooting_request(s.session, "foe:shooter")
    rows = [row for row in weapon_rows(r) if row["wargear_id"] == f"{LORD}:bolt-pistol"]
    assert len(rows) == 1
    status = s.session.submit_parameterized_payload(
        request_id=r.request_id,
        payload=shooting_payload(r, rows, ("own:reactor",)),
        result_id=f"{r.request_id}:incoming",
    )
    assert status.status_kind is not LifecycleStatusKind.INVALID, status
    finish_attacks(s.session)
    hits = [
        cast(dict[str, Any], e["payload"])
        for e in events(s.session, "attack_sequence_step")
        if e["step"] == "hit"
    ]
    assert len(hits) == 1
    assert hits[0]["successful"] is hit
    assert s.state.battlefield_state is not None
    before = s.state.battlefield_state.unit_placement_by_id("own:reactor")
    r = _pending(s)
    if hit:
        _assert_normal_reaction(s, r, "fade_back")
        if accept:
            _accept_normal_reaction(s)
        else:
            request(_choice(s, "decline_triggered_movement"), "select_shooting_unit")
            assert s.state.battlefield_state.unit_placement_by_id("own:reactor") == before
            assert _battle_focus_spends(s) == []
    else:
        assert r.decision_type == "select_shooting_unit"
        assert events(s.session, "shooting_end_surge_triggered") == []
        assert s.state.battlefield_state.unit_placement_by_id("own:reactor") == before
        assert _battle_focus_spends(s) == []
    # The remaining enemy activation can continue lawfully even without a target.
    assert s.session.lifecycle.pending_decision_request() is not None
    r = shooting_request(s.session, "foe:spare")
    assert r.decision_type == "select_charging_unit"


@pytest.mark.parametrize(
    ("maneuver", "history"),
    [
        (m, h)
        for m in ("opportunity_seized", "fade_back")
        for h in ("unit_used", "maneuver_used", "exhausted", "last_token", "later_round")
    ]
    + [("opportunity_seized", "earlier_turn")],
)
def test_native_battle_focus_availability_respects_carried_history(
    faction_definitions: Section02Definitions,
    maneuver: str,
    history: str,
) -> None:
    """Explicit prior-spend checkpoint; only the new trigger and consequence run here."""
    from warhammer40k_core.core.ruleset_descriptor import BattlePhaseKind
    from warhammer40k_core.engine.effects import EffectExpiration, PersistingEffect

    d = faction_definitions
    phase = BattlePhase.MOVEMENT if maneuver == "opportunity_seized" else BattlePhase.SHOOTING
    s = scene(
        d,
        (selection(d, ELDRAD, "reactor"), selection(d, AVENGERS, "other")),
        (selection(d, LORD, "shooter"), selection(d, LORD, "spare")),
        active="opponent",
        phase=phase,
        seed="fade:0",
    )
    if history == "later_round":
        s.state.battle_round = 2
    # Preserve exact source semantics: old phase spends consume round tokens,
    # while one-per-unit and one-per-manoeuvre restrictions expire by phase.
    count = 4 if history in {"exhausted", "later_round"} else 3 if history == "last_token" else 1
    prior_rows = [
        ("own:other", "swift_as_the_wind", BattlePhaseKind.MOVEMENT),
        ("own:other", "flitting_shadows", BattlePhaseKind.CHARGE),
        ("own:other", "sudden_strike", BattlePhaseKind.FIGHT),
        ("own:reactor", "swift_as_the_wind", BattlePhaseKind.MOVEMENT),
    ]
    for i in range(count):
        unit_id, prior_maneuver, started_phase = prior_rows[i]
        if history == "unit_used":
            unit_id, prior_maneuver, started_phase = (
                "own:reactor",
                "flitting_shadows",
                BattlePhaseKind(phase.value),
            )
        elif history == "maneuver_used":
            prior_maneuver, started_phase = maneuver, BattlePhaseKind(phase.value)
        elif history == "earlier_turn":
            unit_id = "own:reactor"
        s.state.record_persisting_effect(
            PersistingEffect(
                effect_id=f"prior-battle-focus:{i}",
                source_rule_id="phase17f:phase17e:aeldari:army-rule",
                owner_player_id="owner",
                target_unit_instance_ids=(unit_id,),
                started_battle_round=1,
                started_phase=started_phase,
                expiration=EffectExpiration.end_battle_round(battle_round=1),
                effect_payload={
                    "effect_kind": "aeldari_battle_focus_token_spent",
                    "maneuver": prior_maneuver,
                    "turn_player_id": "opponent"
                    if history in {"unit_used", "maneuver_used"}
                    else "owner",
                    "unit_instance_id": unit_id,
                    "battle_focus_token_cost": 1,
                },
            )
        )
    locate(s, "foe:shooter", (Pose.at(11.5 if phase is BattlePhase.MOVEMENT else 18, 10),))
    anchor(s)
    s.session.advance_until_decision_or_terminal()
    if phase is BattlePhase.MOVEMENT:
        status = _fall_back(s, "foe:shooter")
    else:
        r = shooting_request(s.session, "foe:shooter")
        rows = [row for row in weapon_rows(r) if row["wargear_id"] == f"{LORD}:bolt-pistol"]
        s.session.submit_parameterized_payload(
            request_id=r.request_id,
            payload=shooting_payload(r, rows, ("own:reactor",)),
            result_id=f"{r.request_id}:incoming",
        )
        finish_attacks(s.session)
        hits = [
            cast(dict[str, Any], e["payload"])
            for e in events(s.session, "attack_sequence_step")
            if e["step"] == "hit"
        ]
        assert hits[0]["successful"]
        status = s.session.advance_until_decision_or_terminal()
    r = request(status)
    if history in {"unit_used", "maneuver_used", "exhausted"}:
        assert r.decision_type == (
            "select_movement_unit" if phase is BattlePhase.MOVEMENT else "select_shooting_unit"
        )
        assert len(_battle_focus_spends(s)) == count
    else:
        _assert_normal_reaction(s, r, maneuver)
        _accept_normal_reaction(s)
        # Accepted fourth token is consumed once; earlier phase history cannot block it.
        assert len(_battle_focus_spends(s)) == count + 1
